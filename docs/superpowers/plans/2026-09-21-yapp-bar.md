# Yapp Bar Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `yapp` launches a menu-bar app; ⌥ Space drops a Spotlight-style pill with the metaball avatar, Yapp listens hands-free and acts as you talk, and the bar hides on the hotkey, Escape, or silence.

**Architecture:** The Claude Design bundle (`window.html`, `yapp-avatar.js`, `tokens.css`) is copied 1:1 into `src/yapp/ui/` and hosted by a frameless transparent pywebview window. A `Bar` façade turns Python calls into `window.yapp.*` JavaScript. `BarDisplay` implements the same `Display` protocol the terminal uses, so the existing `Runner` drives the avatar without changes beyond one `thinking()` hook. `Session` replaces the hold loop with a toggle loop (hotkey / Escape / silence). A PyObjC status item gives the menu, and `yapp install-app` writes a `~/Applications/Yapp.app` wrapper so macOS grants permissions to Yapp.

**Tech Stack:** pywebview 6.2 (Cocoa/WebKit, brings PyObjC), pynput GlobalHotKeys, rich theme, existing pipeline.

**Spec:** `docs/superpowers/specs/2026-09-21-yapp-design.md` section 10 (revised).

## Global Constraints

- Same tooling gates as plan 1: ruff, ruff format, mypy --strict on `src` and `tests`, pytest; commit per task with Conventional Commits and the Co-Authored-By trailer.
- UI files under `src/yapp/ui/` are ported from the design project verbatim except: the Google Fonts `<link>` is replaced by local `@font-face`, the idle hint reads `press ⌥ Space to talk`, and a `<meta name="viewport">` is not added. Any other visual change goes through Claude Design first.
- Never call `evaluate_js` from the pywebview main thread inside a callback that blocks; the session runs in the thread `webview.start(func)` provides.
- Default hotkey `<alt>+<space>`; silence timeout 4 s; max session 60 s; all in `Config`.
- `yapp` with no arguments launches the bar app; `yapp dev` is the terminal hold loop.

```python
"""Python -> JavaScript façade for the pill window. Every call becomes one window.yapp.* call."""

from __future__ import annotations

import json
from typing import Any, Protocol


class WindowLike(Protocol):
    def evaluate_js(self, js: str) -> object: ...
    def show(self) -> None: ...
    def hide(self) -> None: ...
    def move(self, x: int, y: int) -> None: ...


def _j(v: Any) -> str:
    # JSON is valid JS; escape "/" so "</script>" can never terminate an inline block.
    return json.dumps(v).replace("/", "\\/")


class Bar:
    def __init__(self, window: WindowLike, width: int = 360, height: int = 120) -> None:
        self.w = window
        self.width = width
        self.height = height
        self.visible = False

    def show_at_top(self, screen_w: int, screen_h: int, screen_x: int = 0, screen_y: int = 0) -> None:
        x = screen_x + (screen_w - self.width) // 2
        y = screen_y + int(screen_h * 0.12)
        self.w.move(x, y)
        self.w.show()
        self.visible = True

    def hide(self) -> None:
        self.w.hide()
        self.visible = False

    def _call(self, fn: str, *args: Any) -> None:
        self.w.evaluate_js(f"window.yapp.{fn}({', '.join(_j(a) for a in args)})")

    def set_state(self, name: str, **opts: Any) -> None:
        self._call("setState", name, opts)

    def transcript(self, locked: str, pending: str, live: bool = True) -> None:
        self._call("setTranscript", locked, pending, live)

    def decision(self, text: str, muted: bool = False) -> None:
        self._call("setDecision", text, muted)

    def level(self, v: float) -> None:
        self._call("setLevel", round(v, 3))

    def confidence(self, v: float) -> None:
        self._call("setConfidence", round(v, 3))

    def commit(self) -> None:
        self._call("commit")
```

### Task 4: `BarDisplay`: runner hooks to avatar states

**Files:**
- Create: `src/yapp/bardisplay.py`
- Test: `tests/test_bardisplay.py`

**Interfaces:**
- Consumes: `Bar`, `Display` Protocol, `Decision`, `Verdict`, `Result`.
- Produces: `class BarDisplay(bar)` implementing `Display`; plus `begin()` (idle→listening on session start), `end(acted: bool)` (done or unsure), `error_state()`.

State mapping:

| Hook | Bar call |
|---|---|
| `begin()` | `set_state("listening", level=0)`, `decision("")` |
| `listening(level)` | `level(v)` (no state change) |
| `show_transcript(c, p)` | `transcript(" ".join(c), " ".join(p))`; if `len(c)` grew → `commit()` |
| `thinking()` | `set_state("thinking")` |
| `show_decision(d)` | `confidence(d.intent_confidence)` |
| `show_verdict(v)` | EXECUTE → `set_state("acting", direction=DIR[intent])`; WAIT or IGNORE → back to `listening` or `dictating` per `dictating` flag; REFUSE → `set_state("unsure")` + `decision("Won't do that: could delete something")` |
| `show_result(r)` | ok → `decision(text)` where text = message with first letter capitalised ("Opened Notes" → design copy "Opening Notes": use `LABELS` map: opened→"Opening {app}", dictating→"Typing…", quit/erased/closed→"Undone", typed→"Typing…"); not ok → `decision(r.message, muted=True)` |
| `show_error(msg)` | `set_state("error")`, `decision(msg, muted=True)` |
| `status(msg)` | ignored |
| `end(acted)` | `set_state("done")` if acted else `set_state("unsure")` + `decision("Not sure what you meant", muted=True)` |

`DIR = {open_app: 0.0, open_file: 0.0, press_key: -1.2, type_text: 1.4, undo: 3.14}` (radians; right, up-right, down, left) as the `direction` the avatar stretches toward.

```python
"""Runner display hooks -> avatar states and pill text. The runner never knows about the bar."""

from __future__ import annotations

import math
import re

from yapp.bar import Bar
from yapp.types import Decision, Outcome, Result, Verdict

DIRECTION = {"open": 0.0, "press": -1.2, "typed": 1.4, "dictate": 1.4, "undo": math.pi}
REFUSE_COPY = "Won't do that: could delete something"
UNSURE_COPY = "Not sure what you meant"


def result_copy(message: str) -> str:
    m = re.match(r"opened (.+)", message)
    if m:
        return f"Opening {m.group(1)}"
    if message.startswith(("typed", "dictating", "nothing to type")):
        return "Typing…"
    if message.startswith(("quit ", "erased", "closed", "sent cmd+z", "undone")):
        return "Undone"
    if message.startswith("pressed "):
        return message.capitalize()
    return message.capitalize()


class BarDisplay:
    def __init__(self, bar: Bar) -> None:
        self.bar = bar
        self.dictating = False
        self._words = 0
        self.acted = False

    def begin(self) -> None:
        self._words = 0
        self.acted = False
        self.dictating = False
        self.bar.decision("")
        self.bar.transcript("", "")
        self.bar.set_state("listening", level=0.0)

    def end(self, acted: bool) -> None:
        if acted:
            self.bar.set_state("done")
        else:
            self.bar.set_state("unsure")
            self.bar.decision(UNSURE_COPY, muted=True)

    def status(self, msg: str) -> None:
        return None

    def listening(self, level: float) -> None:
        self.bar.level(level)

    def thinking(self) -> None:
        self.bar.set_state("thinking")

    def show_transcript(self, committed: list[str], pending: list[str]) -> None:
        self.bar.transcript(" ".join(committed), " ".join(pending))
        if len(committed) > self._words:
            self.bar.commit()
        self._words = len(committed)

    def show_decision(self, d: Decision) -> None:
        self.bar.confidence(d.intent_confidence)

    def _resting(self) -> None:
        self.bar.set_state("dictating" if self.dictating else "listening")

    def show_verdict(self, v: Verdict) -> None:
        match v.outcome:
            case Outcome.EXECUTE:
                verb = v.reason.split()[0]
                self.acted = True
                self.bar.set_state("acting", direction=DIRECTION.get(verb, 0.0))
            case Outcome.REFUSE:
                self.bar.set_state("unsure")
                self.bar.decision(REFUSE_COPY)
            case _:
                self._resting()

    def show_result(self, r: Result) -> None:
        if r.ok:
            self.bar.decision(result_copy(r.message))
            if r.message == "dictating":
                self.dictating = True
        else:
            self.bar.decision(r.message.capitalize(), muted=True)

    def show_error(self, msg: str) -> None:
        self.bar.set_state("error")
        self.bar.decision(msg, muted=True)
```

`dictating` is also set by `Session` from `runner.stream.dictating` each tick (Task 5) so exits are reflected.

### Task 5: `Session`: the toggle loop

**Files:**
- Create: `src/yapp/session.py`
- Test: `tests/test_session.py`

**Interfaces:**
- Consumes: `Runner`, `Recorder`-like (`arm`, `disarm`, `snapshot`, `level`), `StreamingTranscriber`-like (`update`, `reset`), `BarDisplay`, `Bar`.
- Produces: `class Session(cfg, runner, rec, stt, bar, bardisplay, screen: Callable[[], tuple[int,int,int,int]], clock=time.monotonic, sleep=time.sleep)` with `toggle()` (thread-safe), `escape()`, `run_forever()` (blocks; waits for toggle, runs one session, repeats), `run_one()` (one session, returns `acted: bool`), `stop_requested: bool`.

Session loop (`run_one`):

```
bar.show_at_top(*screen()); bardisplay.begin(); rec.arm(); stt.reset()
last_growth = clock(); words = 0; start = clock()
while not stop_requested:
    t = stt.update(rec.snapshot()); bardisplay.listening(rec.level())
    bardisplay.dictating = runner.stream.dictating
    runner.tick(t.committed, t.pending)
    if len(t.committed) > words: words = len(t.committed); last_growth = clock()
    if clock() - last_growth >= cfg.silence_seconds and words > 0: break
    if clock() - start >= cfg.max_session_seconds: break
    sleep(cfg.tick_seconds)
runner.finish(); rec.disarm()
acted = bardisplay.acted; bardisplay.end(acted)
sleep(cfg.hide_after_seconds); bar.hide(); return acted
```

```python
"""The menu-bar app: pywebview window + global hotkeys + status item + session thread."""

from __future__ import annotations

import threading
from importlib import resources

import webview
from pynput import keyboard

from yapp.audio import Recorder
from yapp.bar import Bar
from yapp.bardisplay import BarDisplay
from yapp.config import Config
from yapp.display import Display, Terminal
from yapp.menubar import install_status_item
from yapp.runner import build_runner
from yapp.session import Session
from yapp.stt import StreamingTranscriber


class Tee:
    """Fan a Display call out to several displays (bar + terminal log)."""

    def __init__(self, *ds: Display) -> None:
        self.ds = ds

    def __getattr__(self, name: str):  # type: ignore[no-untyped-def]
        def call(*a, **k):  # type: ignore[no-untyped-def]
            for d in self.ds:
                getattr(d, name)(*a, **k)
        return call


def _screen_under_mouse() -> tuple[int, int, int, int]:
    from AppKit import NSEvent, NSScreen  # type: ignore[import-not-found]

    p = NSEvent.mouseLocation()
    for s in NSScreen.screens():
        f = s.frame()
        if f.origin.x <= p.x <= f.origin.x + f.size.width and f.origin.y <= p.y <= f.origin.y + f.size.height:
            # Cocoa y grows upward; pywebview wants top-left origin in the primary screen space.
            main_h = NSScreen.screens()[0].frame().size.height
            top = int(main_h - (f.origin.y + f.size.height))
            return int(f.size.width), int(f.size.height), int(f.origin.x), top
    f = NSScreen.mainScreen().frame()
    return int(f.size.width), int(f.size.height), 0, 0


def run_app(cfg: Config, log: bool = False) -> int:
    ui = resources.files("yapp.ui").joinpath("window.html")
    window = webview.create_window(
        "Yapp", url=f"file://{ui}?embed", width=360, height=120, frameless=True,
        transparent=True, on_top=True, resizable=False, shadow=False, easy_drag=False,
        hidden=True, focus=False,
    )
    bar = Bar(window)
    bardisplay = BarDisplay(bar)
    terminal = Terminal() if log else None
    display: Display = Tee(bardisplay, terminal) if terminal else bardisplay  # type: ignore[assignment]

    def session_main() -> None:
        stt = StreamingTranscriber(cfg.whisper_model)
        runner = build_runner(cfg, display)
        rec = Recorder(cfg.sample_rate, cfg.max_hold_seconds)
        rec.start()
        session = Session(cfg, runner, rec, stt, bar, bardisplay, screen=_screen_under_mouse)
        hot = keyboard.GlobalHotKeys({cfg.hotkey_combo: session.toggle, "<esc>": session.escape})
        hot.start()
        install_status_item(
            on_listen=session.toggle,
            on_pause=lambda paused: (hot.stop() if paused else hot.start()),
            on_quit=lambda: (session.quit(), webview.windows[0].destroy()),
            icon_path=str(resources.files("yapp.ui").joinpath("icon.svg")),
        )
        session.run_forever()
        rec.stop()

    webview.start(session_main)
    return 0
```

Escape via `GlobalHotKeys` fires on every Escape system-wide; `session.escape()` only acts while a session is running, so it is harmless otherwise.

```python
"""Write a minimal Yapp.app whose executable execs the project's venv Python."""

from __future__ import annotations

import plistlib
import shutil
import subprocess
import sys
from pathlib import Path

from yapp.config import Config
from yapp.display import Display

PLIST = {
    "CFBundleName": "Yapp",
    "CFBundleDisplayName": "Yapp",
    "CFBundleIdentifier": "co.manali.yapp",
    "CFBundleVersion": "0.1.0",
    "CFBundleShortVersionString": "0.1.0",
    "CFBundlePackageType": "APPL",
    "CFBundleExecutable": "yapp",
    "CFBundleIconFile": "yapp",
    "LSUIElement": True,
    "LSMinimumSystemVersion": "14.0",
    "NSMicrophoneUsageDescription": "Yapp listens while the bar is open so it can act on what you say.",
    "NSInputMonitoringUsageDescription": "Yapp watches for its hotkey (⌥ Space) and Escape.",
    "NSAppleEventsUsageDescription": "Yapp types and presses keys in the app you are using.",
    "NSHighResolutionCapable": True,
}


def _icns(icon_png: Path, dest: Path) -> None:
    """Build an .icns from the 1024 png with iconutil when available, else copy the png."""
    iconset = dest.parent / "yapp.iconset"
    iconset.mkdir(exist_ok=True)
    ok = True
    for size in (16, 32, 128, 256, 512):
        for scale in (1, 2):
            px = size * scale
            name = f"icon_{size}x{size}{'@2x' if scale == 2 else ''}.png"
            r = subprocess.run(["sips", "-z", str(px), str(px), str(icon_png), "--out", str(iconset / name)], capture_output=True)
            ok = ok and r.returncode == 0
    r = subprocess.run(["iconutil", "-c", "icns", str(iconset), "-o", str(dest)], capture_output=True)
    if not (ok and r.returncode == 0):
        shutil.copy(icon_png, dest)
    shutil.rmtree(iconset, ignore_errors=True)


def write_bundle(dest: Path, python: Path, project: Path, icon_png: Path) -> Path:
    macos = dest / "Contents" / "MacOS"
    res = dest / "Contents" / "Resources"
    macos.mkdir(parents=True, exist_ok=True)
    res.mkdir(parents=True, exist_ok=True)
    (dest / "Contents" / "Info.plist").write_bytes(plistlib.dumps(PLIST))
    launcher = macos / "yapp"
    launcher.write_text(
        "#!/bin/zsh\n"
        "source \"$HOME/.zshenv\" 2>/dev/null\n"
        f'cd "{project}"\n'
        f'exec "{python}" -m yapp app "$@"\n'
    )
    launcher.chmod(0o755)
    _icns(icon_png, res / "yapp.icns")
    return dest


def install_app(cfg: Config, display: Display) -> int:
    from importlib import resources

    dest = Path.home() / "Applications" / "Yapp.app"
    if dest.exists():
        shutil.rmtree(dest)
    project = Path(__file__).resolve().parents[2]
    icon = Path(str(resources.files("yapp.ui").joinpath("icon-1024.png")))
    write_bundle(dest, Path(sys.executable), project, icon)
    subprocess.run(["codesign", "--force", "--deep", "--sign", "-", str(dest)], check=False)
    display.status(f"installed {dest}")
    display.status("open it once from Finder; grant Microphone, Input Monitoring, Accessibility to Yapp")
    return 0
```

The launcher sources `~/.zshenv` so `TYPESAFE_API_KEY` reaches the app when launched from Finder.
